% Every call owns its group: parameter changes, coordinates and styles cannot leak.
\newcommand{\DGDeclareDiagram}[2]{%
 \ifcsname DGdiagram-#1\endcsname
  \PackageError{dg-diagrams}{Duplicate diagram '#1'}{Use a unique diagram identifier.}%
 \else\expandafter\gdef\csname DGdiagram-#1\endcsname{#2}\fi}
\newcommand{\DGDiagram}[2][]{%
 \begingroup
 \ifcsname DGdiagram-#2\endcsname
  \begin{tikzpicture}[dg/base,#1]\csname DGdiagram-#2\endcsname\end{tikzpicture}%
 \else\PackageError{dg-diagrams}{Unknown diagram '#2'}{See course-study/figures/catalog.json.}\fi
 \endgroup}
% Primitive API: equal x/y units are preserved unless the caller explicitly changes them.
\newcommand{\DGAxes}[6]{%
 \draw[dg/axis] (#1,0)--(#2,0) node[right] {#5};
 \draw[dg/axis] (0,#3)--(0,#4) node[above] {#6};}
\newcommand{\DGPoint}[3][above right]{%
 \fill[DGInk] (#2) circle (1.4pt) node[#1] {#3};}
\newcommand{\DGAssert}[2]{%
 \pgfmathparse{#2}%
 \ifdim\pgfmathresult pt>0pt\else
  \PackageError{dg-diagrams}{Invalid parameter: #1}{See the parameter conditions in figures/README.md.}%
 \fi}
